## Reproduce

7-gram script (~60 lines, run from a `uv venv` with `pypdf` + `cryptography`):

```python
# /tmp/7gram-check/check.py
import sys, re
from pathlib import Path

STOP = set("the a an and or of in on at to for from with by is are was were "
           "be been being have has had do does did will would shall should "
           "may might must can could".split())

def shingles(text, n=7):
    raw = re.findall(r"[A-Za-z0-9$%][A-Za-z0-9$%'.\-]*", text)
    if len(raw) < n: return {}
    out = {}
    for i in range(len(raw) - n + 1):
        window = raw[i:i+n]
        out.setdefault(" ".join(t.lower() for t in window), []).append(window)
    return out

def nontrivial(window):
    if len([t for t in window if t.lower() not in STOP]) < 4: return False
    has_cap   = any(t[0].isupper() and t.lower() not in STOP for t in window[1:])
    has_digit = any(any(c.isdigit() for c in t) for t in window)
    return has_cap or has_digit

brief = Path(sys.argv[1]).read_text()
bs = shingles(brief)
for ap in sys.argv[2:]:
    a = shingles(Path(ap).read_text())
    common = set(bs) & set(a)
    nt = [(k, bs[k][0]) for k in common if nontrivial(bs[k][0])]
    print(f"{Path(ap).name}: common={len(common)} non-trivial={len(nt)}")
    for k, w in nt[:20]: print(f"  HIT: {' '.join(w)}")
```

PDF extraction: `from pypdf import PdfReader; text = ' '.join(p.extract_text() or '' for p in PdfReader(path).pages)`. BCG India needs `cryptography` (AES). All other anchors extract with `pypdf` alone. Akamai 403s direct on BCG/McKinsey — fetch via Wayback (`https://web.archive.org/web/2024/<orig-url>`).
